3.Stack Demo

package com.collectiondemo;

import java.util.Stack;

public class StackDemo {

	public static void main(String[] args) {
		 Stack<Integer> stack = new Stack<>();

	        // 1. push()
	        // Adds an element to the top of the Stack
	        stack.push(10);
	        stack.push(20);
	        stack.push(30);

	        System.out.println("1. push()  : " + stack);


	        // 2. peek()
	        // Returns the top element without removing it
	        System.out.println("2. peek()  : " + stack.peek());


	        // 3. pop()
	        // Removes and returns the top element
	        int removed = stack.pop();

	        System.out.println("3. pop()   : " + removed);
	        System.out.println("   Stack   : " + stack);


	        // Add another element
	        stack.push(40);
	        stack.push(50);

	        System.out.println("Stack after adding: " + stack);


	        // 4. empty()
	        // Checks whether Stack is empty
	        System.out.println("4. empty() : " + stack.empty());


	        // 5. search()
	        // Returns position of element from the top
	        System.out.println("5. search(40): " + stack.search(40));
	        System.out.println("5. search(10): " + stack.search(10));
	        System.out.println("5. search(100): " + stack.search(100));


	        // Final Stack
	        System.out.println("Final Stack : " + stack);

	}

}

package com.collectiondemo;

import java.util.Vector;

public class VectorExample {

	public static void main(String[] args) {
		Vector<Integer> v = new Vector<>();

        // 1. addElement()
        v.addElement(10);
        v.addElement(20);
        v.addElement(30);
       
        System.out.println("1. addElement()      : " + v);


        // 2. insertElementAt()
        v.insertElementAt(15, 1);
        System.out.println("2. insertElementAt() : " + v);


        // 3. firstElement()
        System.out.println("3. firstElement()    : " + v.firstElement());


        // 4. lastElement()
        System.out.println("4. lastElement()     : " + v.lastElement());


        // 5. elementAt()
        System.out.println("5. elementAt(2)      : " + v.elementAt(2));


        // 6. insertElementAt()
        v.insertElementAt(25, 3);
        System.out.println("6. insertElementAt() : " + v);


        // 7. setElementAt()
        v.setElementAt(100, 2);
        System.out.println("7. setElementAt()    : " + v);


        // 8. removeElementAt()
        v.removeElementAt(2);
        System.out.println("8. removeElementAt() : " + v);


        // 9. removeAllElements()
        Vector<Integer> v2 = new Vector<>();
        v2.addElement(10);
        v2.addElement(20);
        v2.addElement(30);

        v2.removeAllElements();
        System.out.println("9. removeAllElements(): " + v2);


        // 10. capacity()
        System.out.println("10. capacity()       : " + v.capacity());


        // 11. ensureCapacity()
        v.ensureCapacity(50);
        System.out.println("11. ensureCapacity() : " + v.capacity());


        // 12. copyInto()
        Object[] arr = new Object[v.size()];
        v.copyInto(arr);

        System.out.print("12. copyInto()       : ");
        for (Object element : arr) {
            System.out.print(element + " ");
        }
        System.out.println();


        // 13. trimToSize()
        v.trimToSize();
        System.out.println("13. trimToSize()     : " + v.capacity());


        // 14. setSize()
        v.setSize(6);
        System.out.println("14. setSize(6)       : " + v);

	}

}

package com.collectiondemo;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;

public class ComparatorExample {
	
	public static void main(String[] args) {
		List<Integer> list=Arrays.asList(59,54,80,91,62,86);//80 91 62 54 86 59
		
		Comparator<Integer> constomsort=new Comparator<Integer>() {
			@Override
			public int compare(Integer o1, Integer o2) {
				if(o1%10>o2%10) //9>4
					return 1;
				return -1;
			}
		};
		Collections.sort(list);
		Collections.sort(list,constomsort);
		System.out.println(list);
		
		List<Student> stu=new ArrayList<>();
		stu.add(new Student(110,"appu",1200));
		stu.add(new Student(107,"binu",900));
		stu.add(new Student(101,"k7",1190));
		stu.add(new Student(106,"kumar",908));
		stu.add(new Student(103,"aakash",700));
		
		Collections.sort(stu);
		
		for(Student i:stu) {
			System.out.println(i);
		}
	}
}

class Student implements Comparable<Student>{
	int id;
	String name;
	int totalmark;
	public Student(int id, String name, int totalmark) {
		super();
		this.id = id;
		this.name = name;
		this.totalmark = totalmark;
	}
	@Override
	public String toString() {
		return "Student [id=" + id + ", name=" + name + ", totalmark=" + totalmark + "]";
	}
	
	@Override
	public int compareTo(Student o) {
		if(this.totalmark<o.totalmark)
			return 1;
		else
			return -1;
	}
}
