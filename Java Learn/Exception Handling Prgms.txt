		}
		else if(pass.isEmpty()){

		}
	}
}

package com.exceptionhandling;

import java.util.Arrays;

public class BankApplication {

	public static void main(String[] args) {
		BankAccount binu=new BankAccount(3832,"Binu",327453534575l,6838.900);
		System.out.println("Before withdraw amt :"+binu);
		
		try {
			withdraw(binu,-8000);
			System.out.println("After withdraw amt :"+binu);
		} catch (InsuffucientBalanceException e) {
			System.out.println(e);
		} catch (NegativeBalanceException e) {
			System.out.println(e);
		}
		finally {
			System.out.println("Bank Account Logout");
		}
		
	}
	
	public static void withdraw(BankAccount ba,double withdrawamt) throws InsuffucientBalanceException,NegativeBalanceException{
		if(ba.getBalance()<withdrawamt)
			throw new InsuffucientBalanceException("invalid entered amt!!...");
		else if(withdrawamt<=0) 
			throw new NegativeBalanceException("Negative value entered!!...");
		else
			ba.setBalance(ba.getBalance()-withdrawamt);
	}
}
class BankAccount{
	private int custid;
	private String name;
	private long accno;
	private double balance;
	
	public BankAccount(int custid, String name, long accno, double balance) {
		super();
		this.custid = custid;
		this.name = name;
		this.accno = accno;
		this.balance = balance;
	}

	public int getCustid() {
		return custid;
	}

	public void setCustid(int custid) {
		this.custid = custid;
	}

	public String getName() {
		return name;
	}

	public void setName(String name) {
		this.name = name;
	}

	public long getAccno() {
		return accno;
	}

	public void setAccno(long accno) {
		this.accno = accno;
	}

	public double getBalance() {
		return balance;
	}

	public void setBalance(double balance) {
		this.balance = balance;
	}

	@Override
	public String toString() {
		return "BankAccount [custid=" + custid + ", name=" + name + ", accno=" + accno + ", balance=" + balance + "]";
	}
}
class InsuffucientBalanceException extends Exception{
	String msg;

	public InsuffucientBalanceException(String msg) {
		this.msg = msg;
	}

	@Override
	public String toString() {
		return "InsuffucientBalanceException [msg=" + msg + "]";
	}
	
}
class NegativeBalanceException extends Exception{
	String msg;

	public NegativeBalanceException(String msg) {
		this.msg = msg;
	}

	@Override
	public String toString() {
		return "NegativeBalanceException [msg=" + msg + "]";
	}
	
}
